export const syndromeCategories = [
  "Tất cả",

  ...Array.from(
    new Set(
      allSyndromes.map(
        (
          item
        ) =>
          item.category
      )
    )
  ),
];

export const categoryCounts =
  Object.fromEntries(
    syndromeCategories
      .filter(
        (
          item
        ) =>
          item !==
          "Tất cả"
      )
      .map(
        (
          category
        ) => [
          category,

          allSyndromes.filter(
            (
              item
            ) =>
              item.category ===
              category
          ).length,
        ]
      )
  );
